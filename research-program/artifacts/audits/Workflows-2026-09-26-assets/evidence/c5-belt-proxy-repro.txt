'use strict';
const { createRateLimitedGithub } = require('[LOCAL_HOME]/.codex/automations/research-program/[LOCAL_WORKSPACE]/Workflows/.github/scripts/github-rate-limited-wrapper.js');
const { checkRateLimitStatus } = require('[LOCAL_HOME]/.codex/automations/research-program/[LOCAL_WORKSPACE]/Workflows/.github/scripts/github-api-with-retry.js');

async function main() {
  const github = {
    rest: {
      rateLimit: {
        get: async () => ({
          data: {
            resources: {
              core: { remaining: 4000, limit: 5000, reset: 1_700_000_000 },
              graphql: { remaining: 4000, limit: 5000, reset: 1_700_000_000 },
            },
          },
        }),
      },
      issues: { get: async () => ({ data: {} }) },
    },
    request: async () => ({}),
    hook: {},
    graphql: async () => ({}),
  };

  const wrapped = await createRateLimitedGithub({ github, core: console });
  try {
    const status = await checkRateLimitStatus(wrapped, {
      threshold: 0,
      reserveFraction: 0.15,
      estimatedCost: 200,
      failOpen: false,
    });
    console.log('OK safe=', status.safe, 'pool=', status.credentialPoolId);
  } catch (e) {
    console.log('ERR:', e.message);
    process.exitCode = 1;
  }
}

main();
