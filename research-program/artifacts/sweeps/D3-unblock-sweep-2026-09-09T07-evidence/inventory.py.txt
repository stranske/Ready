import concurrent.futures,datetime,json,pathlib,re,subprocess
root=pathlib.Path('/Users/teacher/.codex/automations/research-program')
unit='D3-unblock-sweep-2026-09-09T07'; out=root/'artifacts/sweeps'/f'{unit}-evidence'
repos=re.search(r'SUPPORTED_REPOS=\(([^)]*)\)',pathlib.Path('/Users/teacher/.codex/bin/handoff.sh').read_text()).group(1).split()
repos=[r for r in repos if r!='Orchestrator']
def gh(args):
 p=subprocess.run(['gh',*args],cwd='/tmp',capture_output=True,text=True,timeout=100)
 if p.returncode: raise RuntimeError(p.stderr)
 return json.loads(p.stdout)
def get(repo):
 full='stranske/'+repo
 specs={'repo':['repo','view',full,'--json','defaultBranchRef,url'], 'issues':['issue','list','-R',full,'--state','open','--limit','1000','--json','number,title,labels,url'], 'prs':['pr','list','-R',full,'--state','open','--limit','1000','--json','number,title,labels,updatedAt,isDraft,url']}
 with concurrent.futures.ThreadPoolExecutor(max_workers=3) as pool:data=dict(zip(specs,pool.map(gh,specs.values())))
 branch=data['repo']['defaultBranchRef']['name']
 data['head']=gh(['api',f'repos/{full}/commits/{branch}'])['sha']
 data['runs']=gh(['run','list','-R',full,'--branch',branch,'--limit','5','--json','databaseId,workflowName,status,conclusion,createdAt,headSha,url'])
 workflows=gh(['api',f'repos/{full}/actions/workflows?per_page=100'])
 data['ci']=[]
 for w in workflows['workflows']:
  if w['state']=='active' and (w['name']=='CI' or w['name'].startswith('Gate')):
   runs=gh(['api',f"repos/{full}/actions/workflows/{w['id']}/runs?branch={branch}&per_page=1"])
   data['ci'].append({'name':w['name'],'runs':[{k:r[k] for k in ['id','status','conclusion','created_at','head_sha','html_url']} for r in runs['workflow_runs']]})
 blocked={'needs-human','agents:pause','agents:paused','status:in-progress','tracker:durable','dependencies','epic'}
 data['supply']=sum(not(blocked & {x['name'] for x in i['labels']}) for i in data['issues'])
 frozen=[i for i in data['issues'] if {'needs-human','agents:auto-pilot-pause'} & {x['name'] for x in i['labels']}]
 data['frozen']=[gh(['issue','view',str(i['number']),'-R',full,'--json','number,title,body,author,comments,labels,url']) for i in frozen]
 now=datetime.datetime.now(datetime.timezone.utc)
 data['stalled']=[p for p in data['prs'] if any(x['name'].startswith('agent:') for x in p['labels']) and (now-datetime.datetime.fromisoformat(p['updatedAt'].replace('Z','+00:00'))).total_seconds()>14400]
 data['observedAt']=now.isoformat()
 (out/f'{repo}.json').write_text(json.dumps(data,indent=2)+'\n')
 with (root/'artifacts/sweeps/CHECKPOINT.md').open('a') as f:f.write(f"\n- {data['observedAt']} {unit}: {repo} fresh inventory saved; supply {data['supply']}, frozen {len(frozen)}, stalled {len(data['stalled'])}; detailed CI disposition pending in {out.name}/{repo}.json.\n")
 return {k:data[k] for k in ['supply','stalled','ci','head']}|{'repo':repo,'frozen':[{'number':i['number'],'title':i['title']} for i in frozen]}
with concurrent.futures.ThreadPoolExecutor(max_workers=4) as pool:
 for result in pool.map(get,repos):print(json.dumps(result),flush=True)
